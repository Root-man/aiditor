3/// Times `operation`, doubling the batch until one takes 10 ms, then reports the median.
4pub fn measure(name: &str, mut operation: impl FnMut(), output: &mut String) {
5 let mut iterations = 1;
6 loop {
7 let start = Instant::now();
8 for _ in 0..iterations {
9 operation();
10 }
11 if start.elapsed() >= Duration::from_millis(10) {
12 break;
13 }
14 iterations *= 2;
15 }
16 let mut samples = Vec::new();
17 for _ in 0..21 {
18 let start = Instant::now();
19 for _ in 0..iterations {
20 operation();
21 }
22 samples.push(start.elapsed().as_secs_f64() * 1e9 / iterations as f64);
23 }
24 samples.sort_by(f64::total_cmp);
25 println!("{name}: {:.0} ns/op (median of 21 batches)", samples[10]);
26 output.push_str(&format!(
27 "{name}\t{}\t{}\t{}\n",
28 samples[10], samples[0], samples[20]
29 ));
30}
